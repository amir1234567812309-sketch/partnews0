def api(m, p=None):
    try:
        r = requests.post(API + m, json=p or {}, timeout=20).json()
        if not r.get("ok"): log.warning("%s %s", m, r)
        return r
    except Exception as e:
        log.exception(e); return {"ok": False}

def level_of(p):
    name = LEVELS[0][1]
    for need, lab in LEVELS:
        if p >= need: name = lab
    return name

def next_level(p):
    for need, lab in LEVELS:
        if p < need: return need - p, lab
    return 0, LEVELS[-1][1]

def news_score(title, body=""):
    t = (title or "") + " " + (body or "")
    if JUNK.search(t): return -10
    sc = len(IMPORTANT.findall(t)) * 3
    if len(title or "") >= 25: sc += 1
    if any(x in t for x in ("فوری", "مهم", "رسمی", "اعلام کرد")): sc += 2
    return sc

def clean_html(t):
    if not t: return ""
    return re.sub(r"\s+", " ", unescape(re.sub(r"<[^>]+>", " ", t))).strip()

def fetch_one_rss(src, url, c):
    added = 0
    try:
        r = requests.get(url, timeout=15, headers={"User-Agent": "PartNewsBot"})
        r.raise_for_status()
        ch = ET.fromstring(r.content).find("channel")
        items = ch.findall("item") if ch is not None else []
        for item in items[:12]:
            title = clean_html((item.findtext("title") or "").strip())
            desc = clean_html((item.findtext("description") or "").strip())
            if not title: continue
            if c.execute("SELECT id FROM news WHERE title=?", (title,)).fetchone(): continue
            sc = news_score(title, desc)
            if sc < 3: continue
            c.execute(
                "INSERT INTO news(kind,title,body,link,source,score,is_published,created_at) VALUES(?,?,?,?,?,?,1,?)",
                ("pending", title[:300], desc[:500] or title, (item.findtext("link") or "").strip(),
                 src, sc, datetime.now().strftime("%Y-%m-%d %H:%M:%S")))
            added += 1
    except Exception as e:
        log.warning("RSS %s: %s", src, e)
    return added

def classify_news(c):
    rows = c.execute(
        "SELECT id,title,source,score FROM news WHERE is_published=1 AND kind IN ('pending','day','urgent') ORDER BY score DESC, id DESC LIMIT 20"
    ).fetchall()
    if not rows: return
    c.execute("UPDATE news SET kind='day' WHERE kind IN ('urgent','day','pending')")
    c.execute("UPDATE news SET kind='urgent' WHERE id=?", (rows[0]["id"],))
    for r in rows[1:6]:
        c.execute("UPDATE news SET kind='day' WHERE id=?", (r["id"],))
    tops = rows[:4]
    body = "━━━━━━━━━━━━━━\n📌 گزیده مهم‌ترین اخبار\n━━━━━━━━━━━━━━\n\n"
    for i, x in enumerate(tops, 1):
        body += f"{i}️⃣ {x['title']}\n"
    srcs = sorted({x["source"] for x in tops if x["source"]})
    body += "\n━━━━━━━━━━━━━━\n"
    if srcs: body += f"منابع: {'، '.join(srcs)}\n"
    body += f"آخرین بروزرسانی: {datetime.now():%H:%M}"
    ts = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    old = c.execute("SELECT id FROM news WHERE kind='summary' ORDER BY id DESC LIMIT 1").fetchone()
    if old:
        c.execute("UPDATE news SET body=?,created_at=? WHERE id=?", (body, ts, old["id"]))
    else:
        c.execute("INSERT INTO news(kind,title,body,score,is_published,created_at) VALUES(?,?,?,99,1,?)",
                  ("summary", "خلاصه", body, ts))

def scramble(w):
    a = list(w)
    for _ in range(25):
        random.shuffle(a)
        s = "".join(a)
        if s != w: return s
    return w[::-1]

def greet():
    h = datetime.now().hour
    if h < 12: return "صبح بخیر"
    if h < 18: return "ظهر بخیر"
    return "عصر بخیر"

def one_news(kind):
    c = db()
    r = c.execute(
        "SELECT * FROM news WHERE kind=? AND is_published=1 ORDER BY score DESC, id DESC LIMIT 1",
        (kind,)).fetchone()
    c.close()
    return dict(r) if r else None

def today_challenge():
    day = datetime.now().strftime("%Y-%m-%d")
    c = db()
    row = c.execute("SELECT * FROM challenge WHERE day=?", (day,)).fetchone()
    if row:
        c.close(); return dict(row)
    cat = random.choice(list(QUIZ.keys()))
    qs = QUIZ[cat]
    qi = random.randrange(len(qs))
    q = qs[qi][0]
    c.execute("INSERT OR REPLACE INTO challenge(day,cat,qindex,text) VALUES(?,?,?,?)",
              (day, cat, qi, q))
    c.commit(); c.close()
    return {"day": day, "cat": cat, "qindex": qi, "text": q}

def on_msg(m):
    cid = m["chat"]["id"]
    fr = m.get("from") or {}
    uid = fr.get("id")
    name = fr.get("first_name") or ""
    uname = fr.get("username") or ""
    text = (m.get("text") or "").strip()
    if not uid: return
    st = state.get(uid) or {}

    if text == "/cancel":
        state.pop(uid, None)
        send(cid, "عملیات لغو شد. می‌توانید از منو ادامه دهید.", main_kb())
        return

    if text.startswith("/start"):
        state.pop(uid, None)
        ref = None
        p = text.split(maxsplit=1)
        if len(p) > 1 and p[1].startswith("ref_"):
            try: ref = int(p[1].replace("ref_", ""))
            except Exception: pass
        is_new = ensure(uid, name, uname, ref)
        if is_banned(uid):
            send(cid, "دسترسی شما به ربات محدود شده است. در صورت اعتراض به پشتیبانی پیام دهید.")
            return
        ad = get_setting("start_ad", "").strip()
        if ad:
            send(cid, ad)
        if missing_ch(uid):
            gate(cid)
        else:
            if is_new:
                send(cid,
                     f"🎁 به جمع ما خوش آمدید!\n\n"
                     f"به‌عنوان هدیه ورود، {WELCOME_BONUS} امتیاز به حساب شما اضافه شد.\n"
                     f"از منوی زیر می‌توانید استفاده از ربات را شروع کنید.")
            main_menu(cid, name, uid)
        return

    ensure(uid, name, uname)
    if is_banned(uid):
        send(cid, "دسترسی شما محدود شده است.")
        return
    if uid != ADMIN_ID and missing_ch(uid):
        gate(cid); return
    if not spam_ok(uid):
        return

    if text == "/admin" and uid == ADMIN_ID:
        send(cid,
             "⚙️ پنل مدیریت\n\n"
             "از دکمه‌های زیر برای مدیریت کامل ربات استفاده کنید:\n"
             "آمار، پیام همگانی، اخبار، تبلیغ استارت، قفل عضویت، امتیاز، مسدودسازی و پشتیبانی.",
             adm_kb())
        return

    if uid == ADMIN_ID and st.get("m") == "ad_set":
        set_setting("start_ad", text)
        state.pop(uid, None)
        send(cid, "✅ تبلیغ استارت ذخیره شد.\nاز این پس قبل از ورود کاربر، این متن نمایش داده می‌شود.", adm_kb())
        return
    if uid == ADMIN_ID and st.get("m") == "ch_add":
        if "|" not in text:
            send(cid, "فرمت درست نیست.\nمثال:\npartnews | 📢 کانال اخبار")
            return
        un, title = [x.strip() for x in text.split("|", 1)]
        un = un.lstrip("@")
        c = db()
        c.execute("INSERT OR REPLACE INTO channels(username,title) VALUES(?,?)", (un, title or f"@{un}"))
        c.commit(); c.close()
        state.pop(uid, None)
        send(cid, f"✅ کانال @{un} به قفل عضویت اضافه شد.", adm_kb())
        return
    if uid == ADMIN_ID and st.get("m") == "ch_del":
        un = text.strip().lstrip("@")
        c = db()
        c.execute("DELETE FROM channels WHERE username=?", (un,))
        c.commit(); c.close()
        state.pop(uid, None)
        send(cid, f"✅ اگر کانال @{un} وجود داشت حذف شد.", adm_kb())
        return
    if uid == ADMIN_ID and st.get("m") == "give":
        parts = text.split()
        if len(parts) < 2:
            send(cid, "فرمت: آیدی_عددی مقدار_امتیاز\nمثال:\n123456 20")
            return
        try:
            tid, amount = int(parts[0]), int(parts[1])
            pts(tid, amount)
            state.pop(uid, None)
            send(cid, f"✅ {amount} امتیاز به کاربر {tid} اضافه شد.")
            try: send(tid, f"⭐ مدیریت برای شما {amount} امتیاز واریز کرد.\nموجودی: {pts(tid)}")
            except Exception: pass
        except Exception:
            send(cid, "فرمت نامعتبر است.")
        return
    if uid == ADMIN_ID and st.get("m") == "ban":
        try:
            tid = int(text.strip())
            c = db(); c.execute("UPDATE users SET banned=1 WHERE user_id=?", (tid,)); c.commit(); c.close()
            state.pop(uid, None)
            send(cid, f"🚫 کاربر {tid} مسدود شد.", adm_kb())
        except Exception:
            send(cid, "آیدی عددی معتبر بفرستید.")
        return
    if uid == ADMIN_ID and st.get("m") == "unban":
        try:
            tid = int(text.strip())
            c = db(); c.execute("UPDATE users SET banned=0 WHERE user_id=?", (tid,)); c.commit(); c.close()
            state.pop(uid, None)
            send(cid, f"✅ مسدودی کاربر {tid} برداشته شد.", adm_kb())
        except Exception:
            send(cid, "آیدی عددی معتبر بفرستید.")
        return
    if uid == ADMIN_ID and st.get("m") == "news":
        lines = text.split("\n", 1)
        title = lines[0][:200]
        body = lines[1] if len(lines) > 1 else ""
        sc = max(news_score(title, body), 10)
        c = db()
        c.execute(
            "INSERT INTO news(kind,title,body,source,score,is_published,created_at) VALUES(?,?,?,?,?,1,?)",
            (st.get("k", "day"), title, body, "مدیریت", sc, datetime.now().strftime("%Y-%m-%d %H:%M:%S")))
        c.commit(); c.close()
        state.pop(uid, None)
        send(cid, "✅ خبر با موفقیت ثبت و منتشر شد.", adm_kb())
        return

def ask_q(cid, uid):
    c = db()
    st = c.execute("SELECT * FROM quiz_state WHERE user_id=?", (uid,)).fetchone()
    c.close()
    if not st:
        send(cid, "لطفاً یک موضوع مسابقه انتخاب کنید:", quiz_kb()); return
    cat, idx = st["category"], st["qindex"]
    if cat == "mix" or (isinstance(cat, str) and str(cat).startswith("mix")):
        all_q = []
        for k, qs in QUIZ.items():
            for i, item in enumerate(qs):
                all_q.append((k, i, item))
        if idx >= 10:
            send(cid, "این دور مسابقه ترکیبی به پایان رسید.\nمی‌توانید موضوع دیگری انتخاب کنید.", quiz_kb())
            return
        pick = random.choice(all_q)
        qcat, qidx, (q, opts, correct) = pick
        order = list(range(len(opts)))
        random.shuffle(order)
        shuffled = [opts[i] for i in order]
        correct_pos = order.index(correct)
        c = db()
        c.execute("UPDATE quiz_state SET correct_pos=?, category=? WHERE user_id=?",
                  (correct_pos, f"mix:{qcat}:{qidx}", uid))
        c.commit(); c.close()
        rows = [[{"text": shuffled[i], "callback_data": f"qa:mix:{idx}:{i}"}] for i in range(len(shuffled))]
        rows.append([{"text": "🔙 بازگشت به منو", "callback_data": "back"}])
        send(cid, f"🎲 مسابقه ترکیبی — سؤال {idx + 1} از ۱۰\n\n{q}", {"inline_keyboard": rows})
        return
    qs = QUIZ.get(cat) or []
    if idx >= len(qs):
        send(cid, "سؤالات این موضوع تمام شد.\nموضوع دیگری را انتخاب کنید یا بعداً برگردید.", quiz_kb())
        return
    q, opts, correct = qs[idx]
    order = list(range(len(opts)))
    random.shuffle(order)
    shuffled = [opts[i] for i in order]
    correct_pos = order.index(correct)
    c = db()
    c.execute("UPDATE quiz_state SET correct_pos=? WHERE user_id=?", (correct_pos, uid))
    c.commit(); c.close()
    rows = [[{"text": shuffled[i], "callback_data": f"qa:{cat}:{idx}:{i}"}] for i in range(len(shuffled))]
    rows.append([{"text": "🔙 بازگشت به منو", "callback_data": "back"}])
    send(cid, f"🧠 سؤال {idx + 1} از {len(qs)}\n\n{q}", {"inline_keyboard": rows})

            st = c.execute("SELECT correct_pos,category FROM quiz_state WHERE user_id=?", (uid,)).fetchone()

                c.execute("UPDATE users SET quiz_ok=quiz_ok+1, quiz_combo=quiz_combo+1 WHERE user_id=?", (uid,))

                gain = QUIZ_REWARD + bonus

                if (u["quiz_ok"] or 0) + 1 >= 10: grant_ach(uid, "quiz10", "🎯 ده پاسخ درست")
                if (u["points"] or 0) + gain >= 100: grant_ach(uid, "rich", "💰 رسیدن به صد امتیاز")
            else:
                c.execute("UPDATE users SET quiz_combo=0 WHERE user_id=?", (uid,))
                if cat == "mix" and st and str(st["category"]).startswith("mix:"):
                    _, qc, qi = st["category"].split(":")
                    q, opts, correct = QUIZ[qc][int(qi)]
                else:

                c.commit(); c.close()
                send(cid, f"❌ پاسخ نادرست بود.\nجواب صحیح: {opts[correct]}\nکمبوی شما صفر شد. دوباره تلاش کنید.")

            return

        c.execute("UPDATE users SET points=points-?, title=? WHERE user_id=?", (price, title, uid))

    if not TOKEN or TOKEN == "YOUR_BOT_TOKEN":
        print("لطفاً TOKEN را تنظیم کنید")
        return
    init_db()
    me = api("getMe")
    if not me.get("ok"):
        print("توکن نامعتبر:", me)
        return
    print("ربات روشن شد")
    fetch_rss_news()
    offset = 0
    while True:
        try:
            if time.time() - last_news_fetch >= NEWS_INTERVAL:
                fetch_rss_news()
            res = requests.get(API + "getUpdates", params={"offset": offset, "timeout": 20}, timeout=25).json()
            if not res.get("ok"):
                time.sleep(1); continue
            for u in res.get("result") or []:
                offset = u["update_id"] + 1
                try:
                    if "callback_query" in u:
                        on_cb(u["callback_query"])
                    elif "message" in u:
                        on_msg(u["message"])
                except Exception as e:
                    log.exception(e)
        except Exception as e:
            log.exception(e)
            time.sleep(2)

if __name__ == "__main__":
    main()
